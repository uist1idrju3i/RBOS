cmake_minimum_required(VERSION 3.20.0)

list(PREPEND EXTRA_CONF_FILE "${CMAKE_CURRENT_SOURCE_DIR}/common.conf")
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(rbos)
zephyr_include_directories(include)

if(NOT DEFINED SYSBUILD)
  message(FATAL_ERROR "This must be built with sysbuild.")
endif()
if(NOT CONFIG_BOARD_BM_NRF54L15DK_NRF54L15_CPUAPP_S115_SOFTDEVICE_MCUBOOT)
  message(FATAL_ERROR "RBOS requires the nRF54L15 DK S115 MCUboot board target.")
endif()
if(NOT CONFIG_MCUBOOT_BOOTLOADER_MODE_FIRMWARE_UPDATER)
  message(FATAL_ERROR "Invalid MCUboot mode selected.")
endif()

include(cmake/mrubyc.cmake)
target_include_directories(app PRIVATE ../src/platform/nrf)
target_sources(app PRIVATE ../src/platform/nrf/backend.c)
target_sources(app PRIVATE ../src/platform/nrf/adc.c)
target_sources(app PRIVATE ../src/platform/nrf/gpio.c)
target_sources(app PRIVATE ../src/platform/nrf/button.c)
target_sources(app PRIVATE src/main.c src/runtime.c src/ble.c src/resources.c src/hal/hal.c)
target_sources(app PRIVATE src/resource_probe.c)
